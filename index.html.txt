// ===== HIGH-END BEAUTY / IMMERSIVE PROCESS GALLERY =====
// Automatically reads the existing PROJECTS data and original image paths.
const beautyGallery = document.getElementById('beautyGallery');
const beautyGalleryImage = document.getElementById('beautyGalleryImage');
const beautyGalleryProject = document.getElementById('beautyGalleryProject');
const beautyGalleryStageName = document.getElementById('beautyGalleryStageName');
const beautyGalleryDots = document.getElementById('beautyGalleryDots');
const beautyGalleryThumbnails = document.getElementById('beautyGalleryThumbnails');
// Curate the thumbnail order to match the reference: pink fashion, soft white, dramatic fire.
const beautySeriesOrder = ['high-end-fashion', 'high-end-white', 'high-end-fire'];
const beautyProjectIndices = PROJECTS
  .map((p, i) => p.subcategory === 'high-end-beauty' ? i : -1)
  .filter(i => i >= 0)
  .sort((a, b) => {
    const rank = (i) => {
      const position = beautySeriesOrder.indexOf(PROJECTS[i].id);
      return position === -1 ? beautySeriesOrder.length : position;
    };
    return rank(a) - rank(b);
  });
let beautyGalleryState = { projectIndex: beautyProjectIndices[0], imageIndex: 0, stageIndex: 0 };
let beautyGalleryTrigger = null;

function openBeautyGallery(trigger) {
  if (!beautyProjectIndices.length) return;
  beautyGalleryTrigger = trigger || document.activeElement;
  beautyGalleryState = { projectIndex: beautyProjectIndices[0], imageIndex: 0, stageIndex: 0 };
  beautyGallery.classList.add('open');
  beautyGallery.setAttribute('aria-hidden', 'false');
  document.body.style.overflow = 'hidden';
  renderBeautyGallery();
  document.getElementById('beautyGalleryClose').focus();
}
function closeBeautyGallery() {
  beautyGallery.classList.remove('open');
  beautyGallery.setAttribute('aria-hidden', 'true');
  document.body.style.overflow = '';
  if (beautyGalleryTrigger && typeof beautyGalleryTrigger.focus === 'function') {
    beautyGalleryTrigger.focus();
  }
}
function renderBeautyGallery() {
  const { projectIndex, imageIndex, stageIndex } = beautyGalleryState;
  const project = PROJECTS[projectIndex];
  const series = project.images[imageIndex];
  const stages = series.stages;
  const stage = stages[stageIndex];
  beautyGalleryImage.src = stage.src;
  beautyGalleryImage.alt = `${project.title} – ${stage.label}`;
  beautyGalleryProject.textContent = project.title;
  beautyGalleryStageName.textContent = `${stage.label} / ${String(stageIndex + 1).padStart(2, '0')} OF ${String(stages.length).padStart(2, '0')}`;
  beautyGalleryDots.innerHTML = stages.map((s, i) => `
    <button type="button" class="beauty-gallery-dot" data-beauty-stage="${i}"
      aria-label="${s.label}, stage ${i + 1} of ${stages.length}"
      aria-pressed="${i === stageIndex}" title="${s.label}"></button>
  `).join('');
  beautyGalleryThumbnails.innerHTML = beautyProjectIndices.map((idx, i) => {
    const p = PROJECTS[idx];
    const last = p.images[0].stages.at(-1);
    return `<button type="button" class="beauty-gallery-thumb" data-beauty-project="${idx}"
      aria-label="View ${p.title}" aria-pressed="${idx === projectIndex}" title="${p.title}">
      <img src="${p.thumbnail || last.src}" alt="" style="object-position:${p.thumbPosition || 'center 20%'}">
      <span class="beauty-gallery-thumb-num" aria-hidden="true">0${i + 1}</span>
    </button>`;
  }).join('');
}
beautyGalleryDots.addEventListener('click', (e) => {
  const button = e.target.closest('[data-beauty-stage]');
  if (!button) return;
  beautyGalleryState.stageIndex = Number(button.dataset.beautyStage);
  renderBeautyGallery();
  beautyGalleryDots.querySelector(`[data-beauty-stage="${beautyGalleryState.stageIndex}"]`)?.focus();
});
beautyGalleryThumbnails.addEventListener('click', (e) => {
  const button = e.target.closest('[data-beauty-project]');
  if (!button) return;
  beautyGalleryState.projectIndex = Number(button.dataset.beautyProject);
  beautyGalleryState.imageIndex = 0;
  beautyGalleryState.stageIndex = 0;
  renderBeautyGallery();
  beautyGalleryThumbnails.querySelector(`[data-beauty-project="${beautyGalleryState.projectIndex}"]`)?.focus();
});
document.getElementById('beautyGalleryClose').addEventListener('click', closeBeautyGallery);

function zoomBeautyImage() {
  const { projectIndex, imageIndex, stageIndex } = beautyGalleryState;
  openLightbox(projectIndex, imageIndex);
  if (stageIndex !== 0) goToStage(stageIndex);
}
document.getElementById('beautyGalleryZoom').addEventListener('click', zoomBeautyImage);
beautyGalleryImage.addEventListener('click', zoomBeautyImage);

function getWorkStages(photo){

const lightboxOverlay = document.getElementById('lightboxOverlay');
const lightboxContainer = document.getElementById('lightboxContainer');
const lightboxSteps = document.getElementById('lightboxSteps');
const lightboxHint = document.getElementById('lightboxHint');
const lightboxCaption = document.getElementById('lightboxCaption');
const lightboxZoomHint = document.getElementById('lightboxZoomHint');
let lbState = { stages: [], current: 0 };
let zoomActive = false;

const LOUPE_ZOOM = 2.5;
const ZOOM_ICON_SVG = `<svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.2" stroke-linecap="round" stroke-linejoin="round"><circle cx="10.5" cy="10.5" r="6.5"></circle><line x1="19" y1="19" x2="15.1" y2="15.1"></line></svg>`;
const supportsHover = window.matchMedia('(hover: hover) and (pointer: fine)').matches;

function openLightbox(projectIdx, imgIdx) {
  const img = PROJECTS[projectIdx].images[imgIdx];
  const stages = img.stages;

  lbState = { stages, current: 0 };
  zoomActive = false;

  lightboxContainer.innerHTML = stages.map((s, i) => `
    <img id="lbImg${i}" class="${i === 0 ? 'img-visible' : 'img-hidden'}" src="${s.src}" alt="${s.label}">
  `).join('') + `
    <div class="lightbox-badge before" id="lbBadge">${stages[0].label}</div>
    <div class="lb-loupe" id="lbLoupe"></div>
    <button class="lightbox-zoom-toggle" id="lbZoomToggle" aria-label="Toggle zoom loupe">${ZOOM_ICON_SVG}</button>
  `;

  lightboxCaption.textContent = img.caption || '';
  lightboxHint.textContent = stages.length > 1 ? 'Select a step below to compare' : '';
  lightboxZoomHint.textContent = supportsHover
    ? 'Click the magnifier, then hover the photo to inspect detail'
    : 'Tap the magnifier, then drag on the photo to inspect detail';

  renderLightboxSteps();
  lightboxOverlay.classList.add('open');

  document.getElementById('lbZoomToggle').addEventListener('click', (e) => {
    e.stopPropagation();
    zoomActive = !zoomActive;
    const toggleBtn = document.getElementById('lbZoomToggle');
    toggleBtn.classList.toggle('active', zoomActive);
    if (!zoomActive) {
      const loupeEl = document.getElementById('lbLoupe');
      if (loupeEl) loupeEl.style.display = 'none';
    }
  });
}

function renderLightboxSteps() {
  const { stages, current } = lbState;
  lightboxSteps.innerHTML = stages.map((s, i) => {
    if (s.color) {
      return `<button class="lightbox-step-btn swatch-btn ${i === current ? 'active' : ''}" data-step="${i}" title="${s.label}" style="--swatch-color: ${s.color}"><span class="color-swatch-dot" style="background:${s.color}"></span></button>`;
    }
    return `<button class="lightbox-step-btn ${i === current ? 'active' : ''}" data-step="${i}">${s.label}</button>`;
  }).join('');

  lightboxSteps.querySelectorAll('.lightbox-step-btn').forEach(btn => {
    btn.addEventListener('click', () => goToStage(parseInt(btn.dataset.step)));
  });
}

function goToStage(index) {
  const { stages, current } = lbState;
  if (index === current) return;

  const curImg = document.getElementById(`lbImg${current}`);
  const nextImg = document.getElementById(`lbImg${index}`);
  const badge = document.getElementById('lbBadge');
  if (!curImg || !nextImg || !badge) return;

  curImg.className = 'img-hidden';
  nextImg.className = 'img-visible';

  let badgeClass = 'mid';
  if (index === 0) badgeClass = 'before';
  else if (index === stages.length - 1) badgeClass = 'after';
  badge.className = `lightbox-badge ${badgeClass}`;
  badge.textContent = stages[index].label;

  lbState.current = index;
  renderLightboxSteps();
}

// ===== ZOOM LOUPE =====
function updateLoupe(clientX, clientY) {
  const loupeEl = document.getElementById('lbLoupe');
  const activeImg = lightboxContainer.querySelector('img.img-visible');
  if (!loupeEl || !activeImg) return;

  const containerRect = lightboxContainer.getBoundingClientRect();
  const imgRect = activeImg.getBoundingClientRect();
  const x = clientX - imgRect.left;
  const y = clientY - imgRect.top;

  if (x < 0 || y < 0 || x > imgRect.width || y > imgRect.height) {
    loupeEl.style.display = 'none';
    return;
  }

  const loupeSize = 160;
  loupeEl.style.display = 'block';
  loupeEl.style.left = `${clientX - containerRect.left - loupeSize / 2}px`;
  loupeEl.style.top = `${clientY - containerRect.top - loupeSize / 2}px`;
  loupeEl.style.backgroundImage = `url('${activeImg.src}')`;
  loupeEl.style.backgroundSize = `${imgRect.width * LOUPE_ZOOM}px ${imgRect.height * LOUPE_ZOOM}px`;
  loupeEl.style.backgroundPosition = `${-(x * LOUPE_ZOOM - loupeSize / 2)}px ${-(y * LOUPE_ZOOM - loupeSize / 2)}px`;
}

lightboxContainer.addEventListener('mousemove', (e) => {
  if (!zoomActive) return;
  updateLoupe(e.clientX, e.clientY);
});
lightboxContainer.addEventListener('mouseleave', () => {
  const loupeEl = document.getElementById('lbLoupe');
  if (loupeEl) loupeEl.style.display = 'none';
});

lightboxContainer.addEventListener('touchmove', (e) => {
  if (!zoomActive) return;
  const t = e.touches[0];
  if (!t) return;
  e.preventDefault();
  updateLoupe(t.clientX, t.clientY);
}, { passive: false });
lightboxContainer.addEventListener('touchend', () => {
  const loupeEl = document.getElementById('lbLoupe');
  if (loupeEl) loupeEl.style.display = 'none';
});

// Escape key closes overlays
document.addEventListener('keydown', (e) => {
  if (e.key === 'Escape') {
    if (visualImageViewer.classList.contains('open')) {
      closeVisualImageViewer();
    } else if (lightboxOverlay.classList.contains('open')) {
      lightboxOverlay.classList.remove('open');
    } else if (workGallery.classList.contains('open')) {
      closeWorkGallery();
    } else if (beautyGallery.classList.contains('open')) {
      closeBeautyGallery();
    } else if (visualDirectory.classList.contains('open')) {
      closeVisualDirectory();
    } else if (modalOverlay.classList.contains('open')) {
      closeModal();
    }
  } else if (visualImageViewer.classList.contains('open')) {
    if (e.key === 'ArrowLeft' || e.key === 'ArrowRight') {
      e.preventDefault();
      shiftVisualImage(e.key === 'ArrowRight' ? 1 : -1);
    }
  } else if (workGallery.classList.contains('open') && !lightboxOverlay.classList.contains('open')) {
    const idx=getWorkProjectIndex(), project=PROJECTS[idx];
    if ((e.key === 'ArrowLeft' || e.key === 'ArrowRight') && project.images.length) {
      e.preventDefault();
      workGalleryState.imageIndex=(workGalleryState.imageIndex+(e.key==='ArrowRight'?1:-1)+project.images.length)%project.images.length;
      renderWorkGallery();
    }
  } else if (beautyGallery.classList.contains('open') && !lightboxOverlay.classList.contains('open')) {
    // Arrow keys compare editing stages; up/down jumps between Beauty series.
    if (e.key === 'ArrowLeft' || e.key === 'ArrowRight') {
      e.preventDefault();
      const steps = PROJECTS[beautyGalleryState.projectIndex].images[beautyGalleryState.imageIndex].stages;
      const delta = e.key === 'ArrowRight' ? 1 : -1;
      beautyGalleryState.stageIndex = (beautyGalleryState.stageIndex + delta + steps.length) % steps.length;
      renderBeautyGallery();
    } else if (e.key === 'ArrowUp' || e.key === 'ArrowDown') {
      e.preventDefault();
      const i = beautyProjectIndices.indexOf(beautyGalleryState.projectIndex);
      const delta = e.key === 'ArrowDown' ? 1 : -1;
      beautyGalleryState.projectIndex = beautyProjectIndices[(i + delta + beautyProjectIndices.length) % beautyProjectIndices.length];
      beautyGalleryState.stageIndex = 0;
      beautyGalleryState.imageIndex = 0;
      renderBeautyGallery();
    }
  }
});
